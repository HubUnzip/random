local Admins = {

local Frozen = false
local OriginalSettings = nil

local function getCharacter()
    local character = localPlayer.Character
    if not character then return nil, nil end
    return character, character:FindFirstChildOfClass("Humanoid")
end

local function freezePlayer()
    local _, humanoid = getCharacter()
    if not humanoid then return end
    if not Frozen then
        OriginalSettings = {
            WalkSpeed = humanoid.WalkSpeed,
            UseJumpPower = humanoid.UseJumpPower,
            JumpPower = humanoid.JumpPower,
            JumpHeight = humanoid.JumpHeight,
            AutoRotate = humanoid.AutoRotate
        }
    end
    Frozen = true
    humanoid.WalkSpeed = 0
    humanoid.JumpPower = 0
    humanoid.JumpHeight = 0
    humanoid.AutoRotate = false
end

local function unfreezePlayer()
    local _, humanoid = getCharacter()
    if not humanoid then return end
    if OriginalSettings then
        humanoid.UseJumpPower = OriginalSettings.UseJumpPower
        humanoid.WalkSpeed = OriginalSettings.WalkSpeed
        humanoid.JumpPower = OriginalSettings.JumpPower
        humanoid.JumpHeight = OriginalSettings.JumpHeight
        humanoid.AutoRotate = OriginalSettings.AutoRotate
    else
        humanoid.WalkSpeed = 16
        humanoid.UseJumpPower = true
        humanoid.JumpPower = 50
        humanoid.JumpHeight = 7.2
        humanoid.AutoRotate = true
    end
    Frozen = false
    OriginalSettings = nil
end

local function resetPlayer()
    local character = localPlayer.Character
    if character then
        local humanoid = character:FindFirstChildOfClass("Humanoid")
        if humanoid then humanoid.Health = 0 end
        pcall(function() character:BreakJoints() end)
    end
end

local function teleportToAdmin(admin)
    local character = localPlayer.Character
    local adminCharacter = admin.Character
    local root = character and character:FindFirstChild("HumanoidRootPart")
    local adminRoot = adminCharacter and adminCharacter:FindFirstChild("HumanoidRootPart")
    if root and adminRoot then
        root.CFrame = adminRoot.CFrame * CFrame.new(0, 0, -3)
    end
end

local function handleCommand(admin, message)
    if not admin or not admin:IsA("Player") or not Admins[admin.Name:lower()] then return end
    if typeof(message) ~= "string" then return end
    local command, targetName = message:match("^%;(%S+)%s+(%S+)$")
    if not command or not targetName then return end
    local myName = localPlayer.Name:lower()
    local myDisplay = (localPlayer.DisplayName and localPlayer.DisplayName:lower()) or myName
    local target = targetName:lower()
    if target ~= "all" and target ~= myName and target ~= myDisplay then return end
    command = command:lower()
    if command == "freeze" then freezePlayer()
    elseif command == "unfreeze" then unfreezePlayer()
    elseif command == "reset" then resetPlayer()
    elseif command == "tp" then teleportToAdmin(admin)
    end
end

for _, p in ipairs(Players:GetPlayers()) do
    p.Chatted:Connect(function(msg) handleCommand(p, msg) end)
end
Players.PlayerAdded:Connect(function(p)
    p.Chatted:Connect(function(msg) handleCommand(p, msg) end)
end)

local function get_item_count(item_name)
    local items = localPlayer:FindFirstChild("Items")
    local itemStorage = items and items:FindFirstChild("ItemStorage")
    if not itemStorage then return 0 end

    local target_lower = string.lower(item_name)
    for _, item in ipairs(itemStorage:GetChildren()) do
        local item_name_lower = string.lower(item.Name)
        if item_name_lower == target_lower or string.find(item_name_lower, target_lower, 1, true) then
            local success, val = pcall(function() return item.Value end)
            if success and val ~= nil then
                local num = tonumber(val)
                if num ~= nil then return num end
            end
            for _, child_name in ipairs({"Value", "Amount", "Count", "Quantity"}) do
                local child = item:FindFirstChild(child_name)
                if child and child:IsA("ValueBase") then
                    local num = tonumber(child.Value)
                    if num ~= nil then return num end
                end
            end
            return 1
        end
    end
    return 0
end

local function has_valid_item(item_name)
    return get_item_count(item_name) > 0
end

local function has_weapon(weapon_name)

            if success and val ~= nil then

    if accFolder then
        local direct = accFolder:FindFirstChild(acc_name)
        if direct and direct:IsA("ValueBase") then

            if num and num > 0 then return true end
            if typeof(direct.Value) == "boolean" and direct.Value == true then return true end
        end

        for _, item in ipairs(accFolder:GetChildren()) do
            local item_lower = string.lower(item.Name)
            if item_lower == target_lower or string.find(item_lower, target_lower, 1, true) then

                    if num and num > 0 then return true end
                    if typeof(item.Value) == "boolean" and item.Value == true then return true end

local current_tween = nil
local farmAirPlatform = nil

local function remove_farm_platform()
    if farmAirPlatform then
        pcall(function() farmAirPlatform:Destroy() end)
        farmAirPlatform = nil
    end
    local oldPlat = workspace:FindFirstChild("HeadlessAirPlatform")
    if oldPlat then pcall(function() oldPlat:Destroy() end) end
end

local function get_or_create_farm_platform(root)
    root = root or (localPlayer.Character and localPlayer.Character:FindFirstChild("HumanoidRootPart"))
    if not root then return nil end
    if not farmAirPlatform or not farmAirPlatform.Parent then
        local plat = Instance.new("Part")
        plat.Name = "HeadlessAirPlatform"
        plat.Size = Vector3.new(10, 1, 10)
        plat.Anchored = true
        plat.CanCollide = true
        plat.Transparency = 1
        plat.Material = Enum.Material.SmoothPlastic
        pcall(function() plat.Parent = workspace end)
        farmAirPlatform = plat
    end
    farmAirPlatform.CFrame = root.CFrame * CFrame.new(0, -3.5, 0)
    return farmAirPlatform
end

local function unanchor_character(char)
    char = char or localPlayer.Character
    if not char or Frozen then return end
    for _, part in ipairs(char:GetDescendants()) do
        if part:IsA("BasePart") and part.Anchored then
            part.Anchored = false
        end
    end
end

local function lock_in_place(root)
    local char = localPlayer.Character
    unanchor_character(char)
    root = root or (char and char:FindFirstChild("HumanoidRootPart"))
    if not root then return end
    root.Anchored = false
    root.AssemblyLinearVelocity = Vector3.zero
    root.AssemblyAngularVelocity = Vector3.zero
    get_or_create_farm_platform(root)
    local bv = root:FindFirstChild("HeadlessFreezeVelocity")
    if not bv then
        bv = Instance.new("BodyVelocity")
        bv.Name = "HeadlessFreezeVelocity"
        bv.Velocity = Vector3.zero
        bv.MaxForce = Vector3.new(1e6, 1e6, 1e6)
        bv.Parent = root
    end
end

local function cancel_tween()
    if current_tween then
        current_tween:Cancel()
        current_tween = nil
    end
end

local function unlock_position(root)
    cancel_tween()
    remove_farm_platform()
    local char = localPlayer.Character
    unanchor_character(char)
    root = root or (char and char:FindFirstChild("HumanoidRootPart"))
    if root then
        root.Anchored = false
        local bv = root:FindFirstChild("HeadlessFreezeVelocity")
        if bv then bv:Destroy() end
    end
end

local function set_noclip(character, state)
    if not character then return end
    for _, part in ipairs(character:GetDescendants()) do
        if part:IsA("BasePart") then
            part.CanCollide = not state
        end
    end
end

local function move_to_target(character, root, target_cframe)
    if not (character and root) then return end
    unanchor_character(character)
    unlock_position(root)
    root.Anchored = false
    if Config.MovementMode == "Tween" then
        local distance = (root.Position - target_cframe.Position).Magnitude
        local duration = math.clamp(distance / math.max(Config.TweenSpeed, 10), 0.05, 30)
        local tweenInfo = TweenInfo.new(duration, Enum.EasingStyle.Linear)
        set_noclip(character, true)
        current_tween = TweenService:Create(root, tweenInfo, { CFrame = target_cframe })
        current_tween:Play()
        current_tween.Completed:Wait()
        current_tween = nil
        set_noclip(character, false)
    else
        character:PivotTo(target_cframe)
        root.CFrame = target_cframe
    end
    unanchor_character(character)
    root.Anchored = false
    root.AssemblyLinearVelocity = Vector3.zero
    root.AssemblyAngularVelocity = Vector3.zero
end

local function is_melee(tool)
    if not (tool and tool:IsA("Tool")) then return false end
    local lowerName = string.lower(tool.Name)
    return lowerName == "combat" or lowerName == "baller"
end

local function is_sword(tool)
    if not (tool and tool:IsA("Tool")) then return false end
    local lowerName = string.lower(tool.Name)
    if is_melee(tool) then return false end
    if string.find(lowerName, "blade") or string.find(lowerName, "sword")
        or string.find(lowerName, "katana") or string.find(lowerName, "hammer")
        or string.find(lowerName, "hanger") or string.find(lowerName, "banana") then
        return true
    end
    return false
end

local function equip_weapon()
    local character = localPlayer.Character
    if not character then return nil end
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    local backpack = localPlayer:FindFirstChildOfClass("Backpack")
    if not humanoid then return nil end

    -- Check currently equipped
    for _, item in ipairs(character:GetChildren()) do
        if is_melee(item) or is_sword(item) then
            return item
        end
    end

    -- Equip from backpack
    if backpack then
        -- Prefer Melee
        for _, item in ipairs(backpack:GetChildren()) do
            if is_melee(item) then
                humanoid:EquipTool(item)
                return item
            end
        end
        -- Fallback to Sword
        for _, item in ipairs(backpack:GetChildren()) do
            if is_sword(item) then
                humanoid:EquipTool(item)
                return item
            end
        end
        -- Fallback to any tool
        for _, item in ipairs(backpack:GetChildren()) do
            if item:IsA("Tool") then
                humanoid:EquipTool(item)
                return item
            end
        end
    end

    return character:FindFirstChildOfClass("Tool")
end

        pcall(function() weapon:Activate() end)
    end
end

local auto_skill_cooldowns = {}
local function execute_skills(character, target_pos)
    local tool = character and character:FindFirstChildOfClass("Tool")
    if not tool then return end
    local toolName = tool.Name

    local skillEvent = ReplicatedStorage:FindFirstChild("OtherEvent")
        and ReplicatedStorage.OtherEvent:FindFirstChild("SkillEvents")
        and ReplicatedStorage.OtherEvent.SkillEvents:FindFirstChild("Server_Skills")

    if not (skillEvent and skillEvent:IsA("RemoteEvent")) then return end
    if auto_skill_cooldowns._busy then return end
    auto_skill_cooldowns._busy = true

    task.spawn(function()
        pcall(function()
            for _, key in ipairs({"Z", "X"}) do
                local cdKey = toolName .. "_" .. key
                if os.clock() - (auto_skill_cooldowns[cdKey] or 0) >= 0.4 then
                    auto_skill_cooldowns[cdKey] = os.clock()
                    skillEvent:FireServer(character, toolName, key, "Hold")
                    task.wait(0.08)
                    skillEvent:FireServer(character, toolName, key, "Release", target_pos and { Mouse_Position = target_pos, Hit_Position = target_pos } or nil)
                end
            end
        end)
        auto_skill_cooldowns._busy = false
    end)
end

    end
end

local combat_state = {
    target_mob = nil,
    target_health = nil,
    last_hit_time = os.clock(),
    last_player_health = nil,
    player_hit_count = 0,
    anti_hit_extra_z = 0,
    anti_hit_extra_y = 0,
    adaptive_offset_x = 0,
    reposition_stage = 0
}

local function get_mob_hitbox_info(monster)
    if not monster then return 5, false end
    local maxDim = 5
    pcall(function()
        if monster:IsA("Model") then
            local extents = monster:GetExtentsSize()
            maxDim = math.max(extents.X, extents.Y, extents.Z)
        end
    end)
    local root = monster:FindFirstChild("HumanoidRootPart")
    if root and root:IsA("BasePart") then
        local rootDim = math.max(root.Size.X, root.Size.Y, root.Size.Z)
        if rootDim > maxDim then
            maxDim = rootDim
        end
    end
    local hb = monster:FindFirstChild("Hitbox") or monster:FindFirstChild("AttackHitbox")
    if hb and hb:IsA("BasePart") then
        local hbDim = math.max(hb.Size.X, hb.Size.Y, hb.Size.Z)
        if hbDim > maxDim then
            maxDim = hbDim
        end
    end
    local nameLower = string.lower(monster.Name)
    if string.find(nameLower, "pumpkin") or string.find(nameLower, "lord sus")
        or string.find(nameLower, "evil noob") or string.find(nameLower, "gorilla king")
        or string.find(nameLower, "meme beast") or string.find(nameLower, "gigachad")
        or string.find(nameLower, "mrbeast") or string.find(nameLower, "big floppa") then
        if maxDim < 10 then
            maxDim = 12
        end
    end
    local isBig = (maxDim >= 8)
    return maxDim, isBig
end

local function attack_monster_list(character, player_root, monsters)
    unanchor_character(character)
    if #monsters == 0 then
        current_target_mob = nil
        combat_state.target_mob = nil
        combat_state.target_health = nil
        combat_state.last_player_health = nil
        combat_state.player_hit_count = 0
        combat_state.anti_hit_extra_z = 0
        combat_state.anti_hit_extra_y = 0
        combat_state.adaptive_offset_x = 0
        combat_state.reposition_stage = 0
        return
    end

    local first_mob = monsters[1]
    current_target_mob = first_mob
    local first_mob_root = first_mob:FindFirstChild("HumanoidRootPart")
    local first_mob_hum = first_mob:FindFirstChildOfClass("Humanoid")
    local player_hum = character:FindFirstChildOfClass("Humanoid")

    -- 1. Check if target hitbox is big -> make target always a little farther
    local maxDim, isBig = get_mob_hitbox_info(first_mob)
    local big_extra_z = isBig and math.clamp(maxDim * 0.45, 3.5, 9) or 0
    local big_extra_y = isBig and math.clamp(maxDim * 0.25, 1.5, 5) or 0

    -- 2. Check player hit count: need to be hit 4 times then make monster offset far always a little, if got hit again make far a little more
    if player_hum and player_hum.Health > 0 then
        local current_p_health = player_hum.Health
        if combat_state.last_player_health and current_p_health < combat_state.last_player_health then
            -- Player took damage from monster!
            combat_state.player_hit_count = (combat_state.player_hit_count or 0) + 1

            if combat_state.player_hit_count == 4 then
                -- Hit 4 times: make monster offset far always a little
                combat_state.anti_hit_extra_z = 2.5
                combat_state.anti_hit_extra_y = 1.5
            elseif combat_state.player_hit_count > 4 then
                -- If got hit again: make far a little more each time
                combat_state.anti_hit_extra_z = math.min((combat_state.anti_hit_extra_z or 2.5) + 1.5, 10)
                combat_state.anti_hit_extra_y = math.min((combat_state.anti_hit_extra_y or 1.5) + 1.0, 7)
            end
        end
        combat_state.last_player_health = current_p_health
    end

    -- 3. Check if player cannot hit target in 5 seconds -> change pos so I can hit
    if first_mob_hum and first_mob_hum.Health > 0 then
        local current_m_health = first_mob_hum.Health
        if combat_state.target_mob ~= first_mob then
            combat_state.target_mob = first_mob
            combat_state.target_health = current_m_health
            combat_state.last_hit_time = os.clock()
            combat_state.player_hit_count = 0
            combat_state.anti_hit_extra_z = 0
            combat_state.anti_hit_extra_y = 0
            combat_state.adaptive_offset_x = 0
            combat_state.reposition_stage = 0
        else
            if current_m_health < (combat_state.target_health or current_m_health) then
                -- Hit registered! Reset 5-second timer
                combat_state.target_health = current_m_health
                combat_state.last_hit_time = os.clock()
            else
                local time_without_hit = os.clock() - (combat_state.last_hit_time or os.clock())
                if time_without_hit >= 5 then
                    combat_state.reposition_stage = ((combat_state.reposition_stage or 0) + 1) % 4
                    combat_state.last_hit_time = os.clock()

                    if combat_state.reposition_stage == 1 then
                        combat_state.anti_hit_extra_z = math.max(0, (combat_state.anti_hit_extra_z or 0) - 4)
                        combat_state.anti_hit_extra_y = math.max(0, (combat_state.anti_hit_extra_y or 0) - 2)
                        combat_state.adaptive_offset_x = 0
                    elseif combat_state.reposition_stage == 2 then
                        combat_state.anti_hit_extra_z = 0
                        combat_state.anti_hit_extra_y = 1
                        combat_state.adaptive_offset_x = 2.5
                    elseif combat_state.reposition_stage == 3 then
                        combat_state.anti_hit_extra_z = 0
                        combat_state.anti_hit_extra_y = 1
                        combat_state.adaptive_offset_x = -2.5
                    else
                        combat_state.anti_hit_extra_z = 0
                        combat_state.anti_hit_extra_y = 0
                        combat_state.adaptive_offset_x = 0
                    end

                    if first_mob_root then
                        player_root.CFrame = first_mob_root.CFrame * CFrame.new(0, 6, 7)
                    end
                end
            end
        end
    end

    if first_mob_root then
        local distance = (player_root.Position - first_mob_root.Position).Magnitude
        if distance > 25 then
            move_to_target(character, player_root, first_mob_root.CFrame * CFrame.new(Config.PlayerX, Config.PlayerY, Config.PlayerZ))
            task.wait(0.12)
        end
    end

    lock_in_place(player_root)
    local weapon = equip_weapon()

    -- Calculate safe target position with dynamic offsets
    local target_x = Config.MobDistanceX + (combat_state.adaptive_offset_x or 0)
    local target_y = Config.MobDistanceY + big_extra_y + (combat_state.anti_hit_extra_y or 0)
    local target_z = Config.MobDistanceZ - big_extra_z - (combat_state.anti_hit_extra_z or 0)

    local safe_target_cframe = player_root.CFrame
        * CFrame.new(target_x, target_y, target_z)
        * CFrame.Angles(0, math.rad(180), 0)

    -- Bring & freeze mobs
    for _, monster in ipairs(monsters) do
        local mon_root = monster:FindFirstChild("HumanoidRootPart")
        local mon_humanoid = monster:FindFirstChildOfClass("Humanoid")
        if mon_root and mon_humanoid and mon_humanoid.Health > 0 then
            mon_humanoid.PlatformStand = true
            for _, descendant in ipairs(monster:GetDescendants()) do
                if descendant:IsA("BasePart") then
                    descendant.CanCollide = false
                    descendant.CanTouch = false
                end
            end
            mon_root.CFrame = safe_target_cframe
            mon_root.AssemblyLinearVelocity = Vector3.zero
            mon_root.AssemblyAngularVelocity = Vector3.zero
        end
    end

    auto_attack(weapon)
    if first_mob_root then
        execute_skills(character, first_mob_root.Position)
    end
    trigger_race_v2()
end

local function has_flame_orb()
    if has_valid_item("Flame Orb") then return true end
    local items = localPlayer:FindFirstChild("Items")
    if items then
        for _, c in ipairs(items:GetChildren()) do
            if string.find(string.lower(c.Name), "flame orb") then return true end
        end
    end
    local bp = localPlayer:FindFirstChildOfClass("Backpack")
    if bp then
        for _, c in ipairs(bp:GetChildren()) do
            if string.find(string.lower(c.Name), "flame orb") then return true end
        end
    end
    return false
end

local function trigger_proximity_prompt(prompt)
    if not prompt then return end
    if fireproximityprompt then
        pcall(fireproximityprompt, prompt)
    else
        pcall(function()
            prompt:InputHoldBegin()
            task.wait(prompt.HoldDuration or 0.5)
            prompt:InputHoldEnd()
        end)
    end
end

local function summon_giant_pumpkin(character, player_root)
    set_status("Summoning Giant Pumpkin...")
    unlock_position(player_root)
    local bp = localPlayer:FindFirstChildOfClass("Backpack")
    local hum = character:FindFirstChildOfClass("Humanoid")
    if bp and hum then
        local orbTool = bp:FindFirstChild("Flame Orb")
        if orbTool then pcall(function() hum:EquipTool(orbTool) end) end
    end
    local summonPart = get_pumpkin_summon()
    local statuePos = (summonPart and summonPart:IsA("BasePart")) and (summonPart.CFrame * CFrame.new(0, 3, 0)) or CFrame.new(-1182, -90, 1457)
    move_to_target(character, player_root, statuePos)
    task.wait(0.3)
    local prompt = summonPart and summonPart:FindFirstChildWhichIsA("ProximityPrompt", true)
    if prompt then trigger_proximity_prompt(prompt) end
    task.wait(0.5)
end

local quest_remote = nil
local last_accepted_quest_num = nil
local active_tier_num = nil
local quest_fail_count = 0
local quest_has_reset = false

local function get_quest_remote()
    if quest_remote then return quest_remote end
    local otherEvent = ReplicatedStorage:FindFirstChild("OtherEvent")
    if otherEvent then
        local questEvents = otherEvent:FindFirstChild("QuestEvents")
        if questEvents then
            quest_remote = questEvents:FindFirstChild("Quest")
        end
    end
    return quest_remote
end

local function has_active_quest()
    local pGui = localPlayer:FindFirstChild("PlayerGui")
    if not pGui then return false end
    local questGui = pGui:FindFirstChild("QuestGui")
    if not questGui then return false end
    local holder = questGui:FindFirstChild("Holder")
    if not holder then return false end
    local slot1 = holder:FindFirstChild("QuestSlot1")
    return slot1 and slot1.Visible == true
end

local function abandon_quest()
    local remote = get_quest_remote()
    if remote and remote:IsA("RemoteEvent") then
        pcall(function() remote:FireServer("Abandon_Quest", { QuestSlot = "QuestSlot1" }) end)
        pcall(function() remote:FireServer("Abandon_Quest", "QuestSlot1") end)
    end
    last_accepted_quest_num = nil
end

local function get_tier_info(level)
    for _, tier in ipairs(QUEST_CONFIG) do
        if level >= tier.minLevel then
            return tier
        end
    end
    return QUEST_CONFIG[#QUEST_CONFIG]
end

local function get_quest_island_cframe(tier)
    if not tier then return Island_CFrames.FLOPPA, "floppa" end
    local qNum = tier.questNum or 1
    if qNum <= 6 then
        return Island_CFrames.FLOPPA, "floppa"
    elseif qNum <= 12 then
        return Island_CFrames.SNOW, "snow"
    elseif qNum <= 15 then
        return Island_CFrames.NEW_ISLAND, "new_island"
    elseif qNum <= 19 then
        return Island_CFrames.THIRD, "third"
    elseif qNum <= 23 then
        return Island_CFrames.FOURTH, "fourth"
    elseif qNum == 24 or qNum == 30 or qNum == 31 then
        return Island_CFrames.FIFTH, "fifth"
    elseif qNum >= 25 and qNum <= 28 then
        return Island_CFrames.SIXTH, "sixth"
    elseif qNum == 29 then
        return Island_CFrames.NOOB, "noob"
    elseif qNum >= 32 and qNum <= 36 then
        return Island_CFrames.FORGOTTEN, "forgotten"
    elseif qNum >= 37 and qNum <= 40 then
        return Island_CFrames.SEVENTH, "seventh"
    elseif qNum >= 41 and qNum <= 46 then
        return Island_CFrames.EIGHTH, "eighth"
    end
    return Island_CFrames.FLOPPA, "floppa"
end

local function find_quest_npc(tier)
    local q_folder = workspace:FindFirstChild("NPCs") and workspace.NPCs:FindFirstChild("Quests_Npc")
    if not q_folder then return nil end
    for _, child in ipairs(q_folder:GetChildren()) do
        local name_lower = string.lower(child.Name)
        local is_excluded = (tier.questNum == 1 and (
            string.find(name_lower, "2") or string.find(name_lower, "3") or
            string.find(name_lower, "golden") or string.find(name_lower, "big")
        ))
        if not is_excluded then
            for _, kw in ipairs(tier.npcKeywords) do
                if string.find(name_lower, string.lower(kw), 1, true) then
                    return child
                end
            end
        end
    end
    return nil
end

local function take_specific_quest(questNum)
    if has_active_quest() then
        quest_fail_count = 0
        quest_has_reset = false
        return
    end
    for _, tier in ipairs(QUEST_CONFIG) do
        if tier.questNum == questNum then
            local character = localPlayer.Character
            local player_root = character and character:FindFirstChild("HumanoidRootPart")
            if not (character and player_root) then return end
            unanchor_character(character)
            local npc = find_quest_npc(tier)
            if not npc then
                if questNum == 6 then
                    teleport_to_island(character, player_root, Island_CFrames.FLOPPA, "floppa")
                elseif questNum == 23 then
                    move_to_target(character, player_root, CFrame.new(-1182, -93, 1457))
                elseif questNum == 22 then
                    teleport_to_island(character, player_root, Island_CFrames.FOURTH, "fourth")
                elseif questNum == 29 then
                    teleport_to_island(character, player_root, Island_CFrames.NOOB, "noob")
                elseif questNum == 28 then
                    teleport_to_island(character, player_root, Island_CFrames.SIXTH, "sixth")
                elseif questNum == 32 then
                    teleport_to_island(character, player_root, Island_CFrames.FORGOTTEN, "forgotten")
                elseif questNum == 31 then
                    teleport_to_island(character, player_root, Island_CFrames.FIFTH, "fifth")
                else
                    local targetIsland, islandKey = get_quest_island_cframe(tier)
                    teleport_to_island(character, player_root, targetIsland, islandKey)
                end
                task.wait(0.5)
                npc = find_quest_npc(tier)
            end

            if npc then
                local npc_root = npc:FindFirstChild("HumanoidRootPart") or npc:FindFirstChildWhichIsA("BasePart")
                if npc_root then
                    move_to_target(character, player_root, npc_root.CFrame * CFrame.new(0, 0, -2))
                    task.wait(0.25)
                    local prompt = npc:FindFirstChildWhichIsA("ProximityPrompt", true)
                    if prompt then trigger_proximity_prompt(prompt) end
                    local click = npc:FindFirstChildWhichIsA("ClickDetector", true)
                    if click and fireclickdetector then pcall(fireclickdetector, click) end
                    local remote = get_quest_remote()
                    if remote and remote:IsA("RemoteEvent") then
                        pcall(function() remote:FireServer(npc.Name) end)
                    end
                    last_accepted_quest_num = questNum
                    task.wait(0.5)
                end
            end
            break
        end
    end
end

local function get_quest()
    local character = localPlayer.Character
    if not character then return end
    local player_root = character:FindFirstChild("HumanoidRootPart")
    if not player_root then return end

    unanchor_character(character)

    if has_active_quest() then
        quest_fail_count = 0
        quest_has_reset = false
        return
    end

    local level = get_level()
    local tier = get_tier_info(level)
    local npc = find_quest_npc(tier)

    -- User requirement: make if not found quest 6 then tp to floppa island
    if not npc and tier.questNum == 6 then
        quest_fail_count = quest_fail_count + 1
        set_status("Quest 6 NPC not found -> Teleporting to Floppa Island...", "[STAGE 1/4]")
        teleport_to_island(character, player_root, Island_CFrames.FLOPPA, "floppa")
        task.wait(1)
        return
    end

    -- User requirement: add a check if cannot get quest then try to reset character and if still cant get quest then tp to quest island
    if not npc then
        quest_fail_count = quest_fail_count + 1
        if quest_fail_count >= 3 and not quest_has_reset then
            set_status("Cannot find quest NPC, resetting character...", "[QUEST RECOVERY]")
            quest_has_reset = true
            resetPlayer()
            task.wait(3)
            return
        elseif quest_fail_count >= 5 then
            set_status("Still cannot find quest NPC, TP to quest island...", "[QUEST RECOVERY]")
            local islandCF, islandKey = get_quest_island_cframe(tier)
            teleport_to_island(character, player_root, islandCF, islandKey)
            quest_fail_count = 0
            quest_has_reset = false
            task.wait(1)
            return
        else
            local islandCF, islandKey = get_quest_island_cframe(tier)
            teleport_to_island(character, player_root, islandCF, islandKey)
            task.wait(1)
            return
        end
    end

    local npc_root = npc:FindFirstChild("HumanoidRootPart") or npc:FindFirstChildWhichIsA("BasePart")
    if not npc_root then return end

    move_to_target(character, player_root, npc_root.CFrame * CFrame.new(0, 0, -2))
    task.wait(0.25)
    local prompt = npc:FindFirstChildWhichIsA("ProximityPrompt", true)
    if prompt then trigger_proximity_prompt(prompt) end
    local click = npc:FindFirstChildWhichIsA("ClickDetector", true)
    if click and fireclickdetector then pcall(fireclickdetector, click) end
    local remote = get_quest_remote()
    if remote and remote:IsA("RemoteEvent") then
        pcall(function() remote:FireServer(npc.Name) end)
    end
    task.wait(0.5)

    -- Check if quest was obtained
    if has_active_quest() then
        quest_fail_count = 0
        quest_has_reset = false
    else
        quest_fail_count = quest_fail_count + 1
        -- User requirement: if cannot get quest then try to reset character and if still cant get quest then tp to quest island
        if quest_fail_count >= 3 and not quest_has_reset then
            set_status("Cannot get quest, resetting character...", "[QUEST RECOVERY]")
            quest_has_reset = true
            resetPlayer()
            task.wait(3)
        elseif quest_fail_count >= 5 then
            set_status("Still cannot get quest after reset, TP to quest island...", "[QUEST RECOVERY]")
            local islandCF, islandKey = get_quest_island_cframe(tier)
            teleport_to_island(character, player_root, islandCF, islandKey)
            quest_fail_count = 0
            quest_has_reset = false
            task.wait(1)
        end
    end
end

-- ========================================================================
local function farm_level_step(character, player_root)
    unanchor_character(character)
    local current_lvl = get_level()
    local current_tier = get_tier_info(current_lvl)

    -- Special Quest 23 (Giant Pumpkin / Flame Orb / Quest 22 Scary Skull)
    if current_tier.questNum == 23 or (current_lvl >= 1100 and current_lvl < 1150) then
        local pumpkinBoss = get_boss_or_targets("Giant Pumpkin")
        if #pumpkinBoss > 0 then
            set_status("Killing Giant Pumpkin Boss [Q23]", "[STAGE 1/4] LEVEL FARMING")
            if not has_active_quest() then take_specific_quest(23) end
            attack_monster_list(character, player_root, pumpkinBoss)
            return
        else
            if has_flame_orb() then
                summon_giant_pumpkin(character, player_root)
                return
            else
                set_status("Farming Scary Skull for Flame Orb [Q22]", "[STAGE 1/4] LEVEL FARMING")
                if not has_active_quest() then take_specific_quest(22) end
                local skulls = get_targets("Scary Skull")
                if #skulls > 0 then
                    attack_monster_list(character, player_root, skulls)
                else
                    unlock_position(player_root)
                    move_to_target(character, player_root, CFrame.new(-1434, -93, 1276))
                end
                return
            end
        end
    end

    -- Special Quest 29 (Evil Noob / Noob Head / Quest 28 Moai)
    if current_tier.questNum == 29 or (current_lvl >= 1400 and current_lvl < 1450) then
        local evilNoobBoss = get_boss_or_targets("Evil Noob")
        if #evilNoobBoss > 0 then
            set_status("Killing Evil Noob Boss [Q29]", "[STAGE 1/4] LEVEL FARMING")
            if not has_active_quest() then take_specific_quest(29) end
            attack_monster_list(character, player_root, evilNoobBoss)
            return
        else
            if has_noob_head() then
                summon_evil_noob(character, player_root)
                return
            else
                set_status("Farming Moai for Noob Head [Q28]", "[STAGE 1/4] LEVEL FARMING")
                if not has_active_quest() then take_specific_quest(28) end
                local moais = get_targets("Moai")
                if #moais > 0 then
                    attack_monster_list(character, player_root, moais)
                else
                    unlock_position(player_root)
                    move_to_target(character, player_root, CFrame.new(-2687, -81, 3491))
                end
                return
            end
        end
    end

    -- Special Quest 32 (Lord Sus / Sussy Orb / Quest 31 Sus Duck)
    if current_tier.questNum == 32 or (current_lvl >= 1550 and current_lvl < 1700) then
        local lordSusBoss = get_boss_or_targets("Lord Sus")
        if #lordSusBoss > 0 then
            set_status("Killing Lord Sus Boss [Q32]", "[STAGE 1/4] LEVEL FARMING")
            if not has_active_quest() then take_specific_quest(32) end
            attack_monster_list(character, player_root, lordSusBoss)
            return
        else
            if has_sussy_orb() then
                summon_lord_sus(character, player_root)
                return
            else
                set_status("Farming Sus Duck for Sussy Orb [Q31]", "[STAGE 1/4] LEVEL FARMING")
                if not has_active_quest() then take_specific_quest(31) end
                local ducks = get_targets("Sus Duck")
                if #ducks > 0 then
                    attack_monster_list(character, player_root, ducks)
                else
                    unlock_position(player_root)
                    move_to_target(character, player_root, CFrame.new(1559, -52, 2345))
                end
                return
            end
        end
    end

    -- Standard Tier Quest
    if active_tier_num ~= current_tier.questNum then
        active_tier_num = current_tier.questNum
        if has_active_quest() then
            abandon_quest()
            unlock_position(player_root)
            task.wait(0.3)
        end
    end

    local target_mob = current_tier.mobName
    set_status("Farming " .. target_mob .. " [Quest " .. tostring(current_tier.questNum) .. "]", "[STAGE 1/4] LEVEL FARMING")

    if not has_active_quest() then
        get_quest()
        return
    else
        quest_fail_count = 0
        quest_has_reset = false
    end

    local monsters = get_targets(target_mob)
    if #monsters == 0 then
        unlock_position(player_root)
        if current_tier.questNum <= 6 then
            teleport_to_island(character, player_root, Island_CFrames.FLOPPA, "floppa")
        elseif current_tier.questNum >= 7 and current_tier.questNum <= 12 and current_lvl >= 300 and current_lvl <= 599 then
            teleport_to_island(character, player_root, Island_CFrames.SNOW, "snow")
        end
        return
    end

    attack_monster_list(character, player_root, monsters)
end

local function farm_baller_step(character, player_root)
    if has_baller() then return true end

    local ballCount = get_item_count("Ball")
    local money = get_player_money()

    if ballCount >= 10 and money >= 10000000 then
        set_status("Buying Baller Fighting Style...", "[STAGE 2/4] FIGHT STYLE")
        unlock_position(player_root)
        move_to_target(character, player_root, CFrame.new(1714, -76, -4730))
        task.wait(0.5)
        local sellerEvent = ReplicatedStorage:FindFirstChild("OtherEvent")
            and ReplicatedStorage.OtherEvent:FindFirstChild("MainEvents")
            and ReplicatedStorage.OtherEvent.MainEvents:FindFirstChild("Modules")
        if sellerEvent then
            pcall(function() sellerEvent:FireServer("FightingStyle_Teacher", "Baller") end)
        end
        task.wait(1)
        return has_baller()
    end

    if ballCount < 10 then
        set_status("Farming Baller for Balls (" .. tostring(ballCount) .. "/10)...", "[STAGE 2/4] FIGHT STYLE")
        local ballers = get_targets("Baller")
        if #ballers > 0 then
            attack_monster_list(character, player_root, ballers)
        else
            unlock_position(player_root)
            local distBaller = (player_root.Position - Vector3.new(1301, -54, -4631)).Magnitude
            if distBaller > 20 then
                move_to_target(character, player_root, CFrame.new(1301, -54, -4631))
            end
        end
        return false
    else
        -- Need money
        set_status("Farming Money for Baller ($" .. tostring(money) .. "/$10M)...", "[STAGE 2/4] FIGHT STYLE")
        farm_level_step(character, player_root)
        return false
    end
end

local function is_floppa_quest_active()
    local pGui = localPlayer:FindFirstChild("PlayerGui")
    if not pGui then return false end
    local qGui = pGui:FindFirstChild("QuestGui") or pGui:FindFirstChild("Quest_Gui")
    if not qGui then return false end
    for _, desc in ipairs(qGui:GetDescendants()) do
        if desc:IsA("TextLabel") and desc.Text and string.find(string.lower(desc.Text), "floppa") then
            return true
        end
    end
    return false
end

local function farm_swords_step(character, player_root)
    -- Find first unowned sword in list
    local unownedSword = nil
    for _, swordName in ipairs(SWORD_LIST) do
        local owned = has_weapon(swordName)
        if swordName == "Popcat" and not owned then
            owned = has_weapon("Ohio Popcat")
        end
        if not owned then
            unownedSword = swordName
            break
        end
    end

    if not unownedSword then
        return true -- All swords owned!
    end

    set_status("Farming Sword: " .. unownedSword, "[STAGE 3/4] SWORD FARMING")

    -- 1. Pixel Sword (Obamid drop)
    if unownedSword == "Pixel Sword" then
        local obamids = get_targets("Obamid")
        if #obamids > 0 then
            attack_monster_list(character, player_root, obamids)
        else
            unlock_position(player_root)
            move_to_target(character, player_root, CFrame.new(-494, -61, -3623))
        end
        return false
    end

    -- 2. Pink Hammer (Pink Absorber drop)
    if unownedSword == "Pink Hammer" then
        local absorbers = get_targets("Pink Absorber")
        if #absorbers > 0 then
            attack_monster_list(character, player_root, absorbers)
        else
            unlock_position(player_root)
            move_to_target(character, player_root, CFrame.new(1794, 122, 2993))
        end
        return false
    end

    -- 3. Purple Katana (Lord Sus drop)
    if unownedSword == "Purple Katana" then
        local lordSus = get_boss_or_targets("Lord Sus")
        if #lordSus > 0 then
            attack_monster_list(character, player_root, lordSus)
        else
            if has_sussy_orb() then
                summon_lord_sus(character, player_root)
            else
                local ducks = get_targets("Sus Duck")
                if #ducks > 0 then
                    attack_monster_list(character, player_root, ducks)
                else
                    unlock_position(player_root)
                    move_to_target(character, player_root, CFrame.new(1559, -52, 2345))
                end
            end
        end
        return false
    end

    -- 4. Pumpkin (Nugget Man buy or Giant Pumpkin drop)
    if unownedSword == "Pumpkin" then
        local hasNugget = has_valid_item("Nugget Man")
        local money = get_player_money()
        if hasNugget and money >= 3500000 then
            set_status("Buying Pumpkin Sword from Gravestone...", "[STAGE 3/4] SWORD FARMING")
            unlock_position(player_root)
            move_to_target(character, player_root, CFrame.new(-975, -92, 1284))
            task.wait(0.5)
            local sellerEvent = ReplicatedStorage:FindFirstChild("OtherEvent")
                and ReplicatedStorage.OtherEvent:FindFirstChild("MainEvents")
                and ReplicatedStorage.OtherEvent.MainEvents:FindFirstChild("Modules")
            if sellerEvent then
                pcall(function() sellerEvent:FireServer("Weapon_Seller", "Gravestone") end)
            end
            task.wait(1)
            return false
        end

        local pumpkinBoss = get_boss_or_targets("Giant Pumpkin")
        if #pumpkinBoss > 0 then
            attack_monster_list(character, player_root, pumpkinBoss)
        else
            if has_flame_orb() then
                summon_giant_pumpkin(character, player_root)
            else
                local skulls = get_targets("Scary Skull")
                if #skulls > 0 then
                    attack_monster_list(character, player_root, skulls)
                else
                    unlock_position(player_root)
                    move_to_target(character, player_root, CFrame.new(-1434, -93, 1276))
                end
            end
        end
        return false
    end

    -- 5. Yellow Blade (Evil Noob drop)
    if unownedSword == "Yellow Blade" then
        local evilNoobs = get_boss_or_targets("Evil Noob")
        if #evilNoobs > 0 then
            attack_monster_list(character, player_root, evilNoobs)

-- OPPORTUNISTIC BOSS: MEME BEAST LOOP
-- ========================================================================
local function check_and_kill_meme_beast(character, player_root)
    -- User requirement: wait until player level 1500+ before fighting Meme Beast
    if get_level() < 1500 then
        return false
    end

    local mon_folder = get_monster_folder()
    local memeBeast = mon_folder and mon_folder:FindFirstChild("Meme Beast")
    if memeBeast and memeBeast:FindFirstChild("HumanoidRootPart") then
        local beast_root = memeBeast.HumanoidRootPart
        local beast_hum = memeBeast:FindFirstChildOfClass("Humanoid")
        if not beast_hum or beast_hum.Health > 0 then
            set_status("Killing Meme Beast Event Boss!", "[BOSS EVENT]")
            attack_monster_list(character, player_root, { memeBeast })
            return true
        end
    end
    return false
end

task.spawn(function()
    while true do
        pcall(function()
            local character = localPlayer.Character
            if not character then return end
            local player_root = character:FindFirstChild("HumanoidRootPart")
            local player_humanoid = character:FindFirstChildOfClass("Humanoid")
            if not (player_root and player_humanoid and player_humanoid.Health > 0) then return end

            if check_and_kill_meme_beast(character, player_root) then
                return
            end

            local currentLevel = get_level()

            if currentLevel < Config.MaxLevel then
                farm_level_step(character, player_root)
                return
            end

            -- STAGE 2: Auto Get Fighting Style (Baller)
            if not has_baller() then
                farm_baller_step(character, player_root)
                return
            end

            -- STAGE 3: Auto Farm Swords
            local swordsDone = farm_swords_step(character, player_root)
            if not swordsDone then
                return
            end

            -- STAGE 4: Auto Farm Accessories
            local accsDone = farm_accessories_step(character, player_root)
            if not accsDone then
                return
            end

            -- STAGE 5: All Maxed Out / Endgame Cash Farm
            set_status("All Goals Completed! Farming Cash & Gems", "[MAXED OUT]")
            farm_level_step(character, player_root)
        end)
        task.wait(0.08)
    end
end)

local stats_remote = nil
local function get_stats_remote()
    if stats_remote then return stats_remote end
    local otherEvent = ReplicatedStorage:FindFirstChild("OtherEvent")
    if otherEvent then
        local mainEvents = otherEvent:FindFirstChild("MainEvents")
        if mainEvents then

                if statToggle and Config.AutoStatMelee then
                    remote:InvokeServer({

                        Action = "UpgradeStats",

                    remote:InvokeServer({

                    })

                statToggle = not statToggle

-- Anti Ragdoll / Anti Stun
if Config.AntiRagdoll then
    local function cleanMovers(char)
        if not char then return end
        local root = char:FindFirstChild("HumanoidRootPart")
        if not root then return end
        for _, child in ipairs(root:GetChildren()) do
            if (child:IsA("BodyVelocity") or child:IsA("BodyPosition") or child:IsA("BodyGyro") or child:IsA("LinearVelocity"))
                and child.Name ~= "HeadlessFreezeVelocity" then
                pcall(function() child:Destroy() end)
            end
        end
    end

    local function setupCharStun(char)
        local hum = char:WaitForChild("Humanoid", 5)
        if hum then
            hum.StateChanged:Connect(function(old, new)
                if new == Enum.HumanoidStateType.Ragdoll or new == Enum.HumanoidStateType.FallingDown or new == Enum.HumanoidStateType.PlatformStanding then
                    hum:ChangeState(Enum.HumanoidStateType.GettingUp)
                    cleanMovers(char)
                end
            end)
        end
    end

    if localPlayer.Character then setupCharStun(localPlayer.Character) end
    localPlayer.CharacterAdded:Connect(setupCharStun)
end
